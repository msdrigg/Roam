# Direct dependencies of the helper scripts in this directory. Transitive
# dependencies are intentionally NOT pinned here.
#
# This file used to be a full `pip freeze` dump, which pinned transitives too.
# Dependabot's `scripts-pip` group (.github/dependabot.yml) bumps every pinned
# line independently rather than re-resolving the graph, so it raised
# pydantic_core to 2.48.0 while leaving pydantic at 2.13.4 -- and pydantic
# 2.13.4 requires pydantic-core==2.46.4 exactly. The result was a
# requirements.txt that no resolver could satisfy, which is how scripts/venv
# ended up empty. Listing only direct deps leaves the resolver free to pick
# compatible transitives, so a Dependabot bump can no longer produce an
# unsatisfiable set.
#
#   httpx       -- App Store Connect API calls (sync-metadata, *-tip-iaps,
#                  upload-iap-screenshots, _download_uploaded_screenshots)
#   PyJWT       -- signs the ES256 App Store Connect tokens; the [crypto]
#                  extra pulls in cryptography, which ES256 requires
#   openai      -- localize-auto.py

httpx==0.28.1
PyJWT[crypto]==2.13.0
openai==3.3.1

# Pulled in by PyJWT[crypto], but pinned explicitly so Dependabot keeps
# tracking it -- the lower bound PyJWT declares (>=3.4.0) is open, so bumping
# this line can't conflict the way the old transitive pins did.
cryptography==50.0.1
